# 2. Demonstração do Built-in (cd):

mkdir pasta_teste_shell

pwd

pwd



# 3. Demonstração de Redirecionamento de Saída (>)

echo "Primeira linha do arquivo de teste" > saida.txt

cat saida.txt

cat arquivos.txt



# 4. Demonstração de Concatenação (>> - se implementado)

# 5. Demonstração de Redirecionamento de Entrada (<)

wc -l < arquivos.txt
